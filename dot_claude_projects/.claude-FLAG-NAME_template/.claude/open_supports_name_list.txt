# open_supports_name_list.txt
#
# Each non-empty, non-comment line:
#   <support-name> [install args...]
#
# Accepted support names:
#   OwnerName/RepoName
#   ost_OwnerName_RepoName
#
# {PROJECT_DIR} in args is replaced with the absolute project path. Install
# scripts run inside the vendored package, so project-level installs need it.
# User-level (global) packages belong in dot_claude_projects/.claude/ instead.
#
# Examples:
# colbymchenry/codegraph --target=claude --location=local {PROJECT_DIR}
# open-gsd/gsd-core --claude --local {PROJECT_DIR}
